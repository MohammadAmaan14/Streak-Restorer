```
(async () => {
    const results = new Map();
    const TARGET = 'لقمان~';

    const scan = () => {
        document.querySelectorAll('.O4POs').forEach(row => {
            const name = row.querySelector('[id^="title-"]')?.innerText?.trim();
            const meta = row.querySelector('.ovUsZ');

            if (!name || !meta) return;

            const streakEl = [...meta.querySelectorAll('span')]
                .find(s => /^\d+\s*🔥$/.test(s.innerText.trim()));

            results.set(name, {
                name,
                streak: streakEl ? parseInt(streakEl.innerText) : null,
                hasStreak: !!streakEl
            });
        });
    };

    let foundTarget = false;

    for (let i = 0; i < 100; i++) {
        scan();

        // Check whether target has appeared
        if ([...results.keys()].some(name => name.replace(/\s+/g, '') === TARGET.replace(/\s+/g, ''))) {
            foundTarget = true;
            break;
        }

        const rows = [...document.querySelectorAll('.O4POs')];
        const last = rows.at(-1);

        if (!last) break;

        last.scrollIntoView({
            behavior: 'instant',
            block: 'end'
        });

        await new Promise(r => setTimeout(r, 800));
    }

    scan();

    console.clear();

    const output = [...results.values()];

    console.table(output);

    console.log(
        foundTarget
            ? `✅ Target "${TARGET}" found.`
            : `⚠️ Target "${TARGET}" was not found.`
    );

    console.log(`Total chats loaded: ${output.length}`);
    console.log(`Chats with streaks: ${output.filter(x => x.hasStreak).length}`);
})();
```
